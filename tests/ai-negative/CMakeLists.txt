cmake_minimum_required(VERSION 3.15)
project(AiNegativeTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL 7.62 REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(RT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(ai-negative-tests test_service.cc "${RT_ROOT}/rtgui/ainegative.cc" "${RT_ROOT}/rtengine/cJSON.c")
target_include_directories(ai-negative-tests PRIVATE "${RT_ROOT}")
target_link_libraries(ai-negative-tests PRIVATE CURL::libcurl Threads::Threads)
enable_testing()
add_test(NAME ai-negative-service COMMAND ai-negative-tests)
set_tests_properties(ai-negative-service PROPERTIES TIMEOUT 30)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME ai-negative-http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_http.py" "$<TARGET_FILE:ai-negative-tests>")
    set_tests_properties(ai-negative-http PROPERTIES TIMEOUT 30)
endif()
